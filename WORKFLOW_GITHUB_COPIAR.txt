name: Oitosete FlowText 0.9.4 — Android e iOS

# Execute manualmente para controlar uso de minutos e downloads grandes.
on:
  workflow_dispatch:
    inputs:
      modelo:
        description: 'Reconhecimento: streaming pt-BR (novo) ou Whisper'
        required: true
        default: 'streaming'
        type: choice
        options: [streaming, streaming_leve, base, tiny, small]
      tipo_apk:
        description: 'APK Android: menor para celulares modernos (release_arm64), todas ABIs ou debug'
        required: true
        default: 'release_arm64'
        type: choice
        options: [release_arm64, release_todas, debug]
      plataforma:
        description: 'Compilar Android, iPhone ou ambos (iPhone sem assinatura)'
        required: true
        default: 'android'
        type: choice
        options: [android, ios, ambos]

jobs:
  android:
    if: inputs.plataforma == 'android' || inputs.plataforma == 'ambos'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'
      - uses: subosito/flutter-action@v2
        with:
          channel: stable
          cache: true
      - name: Verificar scripts de identidade e modelo
        run: python3 -m unittest discover -s scripts/tests -v
      - name: Gerar arquivos Android/iOS
        run: flutter create --platforms=android --project-name=teleprompter_ai .
      - name: Configurar permissões, nome, ícone e abertura
        run: |
          python3 scripts/configurar_plataformas.py --platform android
          python3 scripts/configurar_splash.py --platform android
      - name: Instalar modelo de reconhecimento
        shell: bash
        run: |
          case '${{ inputs.modelo }}' in
            streaming)
              # Comportamento anterior: streaming e Whisper Tiny de segurança.
              python3 scripts/instalar_streaming.py
              python3 scripts/instalar_modelo.py --model tiny
              ;;
            streaming_leve)
              # Menor APK: mantém Nemotron; SEM fallback Whisper offline.
              python3 scripts/instalar_streaming.py
              ;;
            tiny|base|small)
              python3 scripts/instalar_modelo.py --model '${{ inputs.modelo }}'
              ;;
            *) exit 2 ;;
          esac
      - name: Dependências e análise
        run: |
          flutter pub get
          rm -f test/widget_test.dart
          flutter analyze --no-fatal-infos --no-fatal-warnings
          flutter test
      - name: Compatibilidade AndroidX Media3
        run: |
          cat >> android/build.gradle.kts <<'EOF'
          allprojects {
              configurations.configureEach {
                  resolutionStrategy.eachDependency {
                      if (requested.group == "androidx.media3") {
                          useVersion("1.9.4")
                      }
                  }
              }
          }
          EOF
      - name: Adaptar native_video_editor ao Media3 1.9.4
        shell: bash
        run: |
          python3 - <<'PYTHON'
          import os
          from pathlib import Path

          cache = Path(os.environ.get('PUB_CACHE') or (Path.home() / '.pub-cache'))
          kotlin = cache / 'hosted' / 'pub.dev' / 'native_video_editor-0.4.0' / 'android' / 'src' / 'main' / 'kotlin' / 'com' / 'example' / 'native_video_editor'
          original = 'EditedMediaItemSequence('
          for filename in ('AudioMergerPipeline.kt', 'ImageComposerPipeline.kt'):
              path = kotlin / filename
              if not path.is_file():
                  raise SystemExit(f'Pacote Kotlin não encontrado: {path}')
              source = path.read_text(encoding='utf-8')
              occurrences = source.count(original)
              migrated_video = source.count('EditedMediaItemSequence.withVideoFrom(')
              migrated_audio = source.count('EditedMediaItemSequence.withAudioFrom(')

              if occurrences == 2 and migrated_video == 0 and migrated_audio == 0:
                  # Primeiro item: vídeo. Segundo item: áudio.
                  source = source.replace(original, 'EditedMediaItemSequence.withVideoFrom(', 1)
                  source = source.replace(original, 'EditedMediaItemSequence.withAudioFrom(', 1)
                  path.write_text(source, encoding='utf-8')
                  print(f'{filename}: duas chamadas migradas para a API Media3 1.9+')
              elif occurrences == 0 and migrated_video == 1 and migrated_audio == 1:
                  print(f'{filename}: já corrigido; nenhuma alteração necessária')
              else:
                  raise SystemExit(
                      f'Estado inesperado em {filename}: antigas={occurrences}, '
                      f'video={migrated_video}, audio={migrated_audio}. '
                      'A correção foi interrompida para evitar alterações indevidas.'
                  )
          PYTHON
      - name: Compilar APK Android no modo solicitado
        shell: bash
        run: |
          case '${{ inputs.tipo_apk }}' in
            release_arm64)
              # Padrão recomendado: um único APK arm64, sem recursos de debug.
              flutter build apk --release --target-platform android-arm64 --split-per-abi \
                --split-debug-info=build/simbolos/android
              ;;
            release_todas)
              # APK separado por arquitetura para dispositivos mais antigos.
              flutter build apk --release --split-per-abi \
                --split-debug-info=build/simbolos/android
              ;;
            debug)
              # Apenas para investigação de falhas; arquivo muito maior.
              flutter build apk --debug
              ;;
            *) exit 2 ;;
          esac
      - name: Medir tamanho real do APK e componentes
        run: python3 scripts/relatorio_tamanho.py
      - uses: actions/upload-artifact@v4
        with:
          name: oitosete-flowtext-v094-${{ inputs.modelo }}-${{ inputs.tipo_apk }}
          path: |
            build/app/outputs/flutter-apk/*.apk
            relatorio_tamanho_build.md
            build/simbolos/android/**
          retention-days: 7
          if-no-files-found: error

  ios:
    if: inputs.plataforma == 'ios' || inputs.plataforma == 'ambos'
    runs-on: macos-15
    steps:
      - uses: actions/checkout@v4
      - uses: subosito/flutter-action@v2
        with:
          channel: stable
          cache: true
      - name: Verificar scripts de identidade e modelo
        run: python3 -m unittest discover -s scripts/tests -v
      - name: Gerar arquivos Android/iOS
        run: flutter create --platforms=ios --project-name=teleprompter_ai .
      - name: Configurar permissões, nome, ícone e abertura
        run: |
          python3 scripts/configurar_plataformas.py --platform ios
          python3 scripts/configurar_splash.py --platform ios
      - name: Instalar modelo de reconhecimento
        shell: bash
        run: |
          case '${{ inputs.modelo }}' in
            streaming)
              # Comportamento anterior: streaming e Whisper Tiny de segurança.
              python3 scripts/instalar_streaming.py
              python3 scripts/instalar_modelo.py --model tiny
              ;;
            streaming_leve)
              # Menor APK: mantém Nemotron; SEM fallback Whisper offline.
              python3 scripts/instalar_streaming.py
              ;;
            tiny|base|small)
              python3 scripts/instalar_modelo.py --model '${{ inputs.modelo }}'
              ;;
            *) exit 2 ;;
          esac
      - name: Dependências e análise
        run: |
          flutter pub get
          rm -f test/widget_test.dart
          flutter analyze --no-fatal-infos --no-fatal-warnings
          flutter test
      - name: Compilar iOS sem assinatura
        run: flutter build ios --release --no-codesign
      - name: Resumo do tamanho iOS (app não assinado)
        shell: bash
        run: |
          du -sh build/ios/iphoneos/Runner.app | tee relatorio_tamanho_ios.txt
          echo '## Tamanho iOS (app não assinado)' >> "$GITHUB_STEP_SUMMARY"
          echo "$(cat relatorio_tamanho_ios.txt)" >> "$GITHUB_STEP_SUMMARY"
      - name: Arquivar aplicativo iOS não assinado
        run: cd build/ios/iphoneos && zip -qry ../../../oitosete-flowtext-ios-sem-assinatura.zip Runner.app
      - uses: actions/upload-artifact@v4
        with:
          name: oitosete-flowtext-v094-${{ inputs.modelo }}-ios-unsigned
          path: |
            oitosete-flowtext-ios-sem-assinatura.zip
            relatorio_tamanho_ios.txt
          retention-days: 7
          if-no-files-found: error
